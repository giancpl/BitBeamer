\newcommand{\studio@closingboolean}[1]{%
  \edef\studio@closingvalue{#1}\def\studio@closingtrue{true}\def\studio@closingfalse{false}%
  \ifx\studio@closingvalue\studio@closingtrue\studio@closingqrtrue\else
    \ifx\studio@closingvalue\studio@closingfalse\studio@closingqrfalse\else
      \PackageError{bitpolito-studio}{Invalid boolean '#1' for closing qrcode}{Use qrcode=true or qrcode=false.}%
    \fi
  \fi
}
\newcommand{\studio@closingimage}[1]{%
  \IfFileExists{#1}{\includegraphics[width=554\studio@closingunit,height=554\studio@closingunit,keepaspectratio]{#1}}{%
    \PackageError{bitpolito-studio}{Missing closing QR image '#1'}{Restore the prepared artwork or supply qr-image with a project-relative image path.}%
  }%
}

% A tracked copy of the current font keeps all other slide typography untouched.
\newcommand{\studio@closingtrack}[1]{%
  \ifdefined\letterspacefont
    \letterspacefont\studio@closingtrackedfont\font #1\relax
    \fontdimen2\studio@closingtrackedfont=\dimexpr\numexpr1000+#1\relax sp*\fontdimen2\studio@closingtrackedfont/1000\relax
    \studio@closingtrackedfont
  \fi
}
% Fixed text boxes report overflow instead of shrinking typography.
\newcommand{\studio@closingtext}[9]{%
  \node[anchor=north west,inner sep=0] at (#2,#3) {%
    \setbox\studio@closingtextbox=\vbox{\hsize=#4\studio@closingunit\parindent=0pt
      \normalfont\sffamily\fontsize{#6\studio@closingunit}{#7\studio@closingunit}\selectfont
      \color{white}\centering#8\noindent#9\par}%
    \ifdim\dimexpr\ht\studio@closingtextbox+\dp\studio@closingtextbox\relax>#5\studio@closingunit
      \PackageError{bitpolito-studio}{Text overflow in closing/#1}{Shorten the text. The closing layout does not shrink fonts.}%
    \fi
    \box\studio@closingtextbox};%
}
