% Each public card groups its own text and optional icon, in reading order.
\newcommand{\studiocardfields}[4]{%
 \pgfkeys{/studio fields/card-#1/.code={\pgfqkeys{/studio card/#1}{##1}},
  /studio card/#1/.is family,
  /studio card/#1/title/.code=\pgfkeyssetvalue{/studio values/#2}{##1},
  /studio card/#1/body/.code=\pgfkeyssetvalue{/studio values/#3}{##1},
  /studio card/#1/.unknown/.code=\PackageError{bitpolitostudiomodels}{Unknown card-#1 field '\pgfkeyscurrentname'}{Use title, body and, where available, icon, image or graphic.}}%
 \def\studiocardiconslot{#4}%
 \ifx\studiocardiconslot\empty\else
  \pgfkeys{/studio card/#1/icon/.style={/studio fields/icon/#4={##1}},
   /studio card/#1/image/.style={/studio fields/image/#4={##1}},
   /studio card/#1/graphic/.style={/studio fields/graphic/#4={##1}}}%
 \fi}
