  \pgfkeysgetvalue{/studio values/quote}\studioquotecopy
  \pgfkeysgetvalue{/studio values/attribution}\studioquotecitation

  \setbox\studiotextbox=\vbox{\hsize=\studioquotetextwidth\studiounit\parindent=0pt

   \color{bitpolito}\raggedright\bfseries
   \ifx\studioquotecopy\empty\else\noindent\textquotedblleft\studioquotecopy\textquotedblright\par\fi}%

  \begin{tikzpicture}[x=\studiounit,y=-\studiounit,every node/.style={inner sep=0,outer sep=0}]

    (\studioquotebackx,\studioquotebacky) rectangle ({\studioquotebackx+\studioquotebackwidth},{\studioquotepanelheight+\studioquotebackbottom});

    \node[anchor=north west] at (97.07,115.316) {\box\studiotextbox};

   \fi

 };
