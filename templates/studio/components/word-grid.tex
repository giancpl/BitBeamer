% Shared twelve-word grid. Panels stay fixed; editable text never shrinks.
\ifdefined\studiowordcount\else\newcount\studiowordcount\fi
\def\studiowordnumbered{true}
\makeatletter
\def\studiowordset#1{%
 \studiowordcount=0
 \@for\studiowordvalue:=#1\do{%
  \advance\studiowordcount by 1
  \ifnum\studiowordcount>12
   \PackageError{bitpolitostudiotemplates}{Word grid requires exactly twelve words}{Provide twelve comma-separated values in reading order.}%
  \else
   \ifx\studiowordvalue\@empty
    \PackageError{bitpolitostudiotemplates}{Empty word in word grid}{Provide twelve nonempty values.}%
   \fi
   \pgfkeyslet{/studio values/word-\the\studiowordcount}\studiowordvalue
  \fi}%
 \ifnum\studiowordcount=12\else
  \PackageError{bitpolitostudiotemplates}{Word grid requires exactly twelve words}{Provide twelve comma-separated values in reading order.}%
 \fi}
\makeatother
\pgfkeys{/studio fields/words/.code=\studiowordset{#1},
 /studio fields/numbered/.is choice,
 /studio fields/numbered/true/.code=\def\studiowordnumbered{true},
 /studio fields/numbered/false/.code=\def\studiowordnumbered{false}}
\studioapplyfields
